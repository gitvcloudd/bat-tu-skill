### 8.2. Giải Quyết Khủng Hoảng Chuyển Giao Quyền Lực Gia Tộc F1 $\to$ F2 `[REF-12-016]`

Thống kê toàn cầu cho thấy hơn 70% doanh nghiệp gia đình sụp đổ ở thế hệ F2. Nguyên nhân cốt lõi dưới lăng kính Tử Bình là **SỰ XUNG ĐỘT KHÍ THẾ GIỮA THỜI KỲ KHAI SƠ PHÁ THẠCH (F1) VÀ THỜI KỲ QUẢN TRỊ THỂ CHẾ (F2)**:
- Người cha F1 thường có Bát tự Thất Sát, Dương Nhận, Thương Quan vượng (tính cách độc đoán, quyết liệt, liều lĩnh mới lập nên cơ nghiệp giữa thời loạn).
- Khi chọn con kế nghiệp, nếu chọn người con có Bát tự y hệt mình (cũng Sát Nhận vượng) thì hai hổ một chuồng, cha con tương tàn xung đột triệt để.
- **Chiến lược lựa chọn F2 chuẩn mực:** Phải chọn người con có cấu trúc **Chính Quan, Chính Ấn, Chính Tài** — người có năng lực thể chế hóa tập đoàn, xây dựng quy trình minh bạch, quản trị bằng dữ liệu và phần mềm, đưa doanh nghiệp gia đình thành công ty đại chúng niêm yết.

## CHƯƠNG XI: ĐỘNG CƠ MÃ NGUỒN TYPESCRIPT CHUẨN DOANH NGHIỆP: grandUnifiedEngine.ts `[REF-12-021]`

### 11.1. Kiến Trúc Hợp Nhất 6 Động Cơ Toán Học

```typescript
export interface BaziProfile {
  ugepScore: number;     // 0 - 100: Sức mạnh Dụng thần
  cbiRaw: number;        // -100 đến +100: Cân bằng nhiệt ẩm Điều hậu
  ppsScore: number;      // 0 - 100: Phẩm cấp Cách cục Bát Chính
  fhiScore: number;      // 0 - 100: Hòa hợp Gia đạo Lục thân
  cpiScore: number;      // 0 - 100: Rủi ro Tích lũy Vận hạn (Trừ)
  wesScore: number;      // 0 - 100: Hiệu suất Tố công Manh phái
}

export interface GufiResult {
  gufiScore: number;
  rankLevel: "Thần Cách" | "Quý Cách" | "Bình Khang" | "Suy Vi" | "Trụy Lạc";
  interventionPriority: string[];
  recommendation: string;
}

export class GrandUnifiedFateEngine {
  private static readonly W_UGEP = 0.15;
  private static readonly W_CBI  = 0.15;
  private static readonly W_PPS  = 0.20;
  private static readonly W_FHI  = 0.15;
  private static readonly W_CPI  = 0.15;
  private static readonly W_WES  = 0.20;

  public static evaluate(profile: BaziProfile): GufiResult {
    // Chuẩn hóa CBI từ [-100, +100] về [0, 100]
    const cbiNorm = Math.max(0, Math.min(100, (profile.cbiRaw + 100) / 2));

    // Tính toán GUFI tổng hợp
    const rawGufi = (
      this.W_UGEP * profile.ugepScore +
      this.W_CBI  * cbiNorm +
      this.W_PPS  * profile.ppsScore +
      this.W_FHI  * profile.fhiScore -
      this.W_CPI  * profile.cpiScore +
      this.W_WES  * profile.wesScore
    );

    const finalGufi = Math.max(0, Math.min(100, Math.round(rawGufi)));

    let rank: GufiResult["rankLevel"] = "Trụy Lạc";
    if (finalGufi >= 90) rank = "Thần Cách";
    else if (finalGufi >= 75) rank = "Quý Cách";
    else if (finalGufi >= 50) rank = "Bình Khang";
    else if (finalGufi >= 30) rank = "Suy Vi";

    const priorities: string[] = [];
    if (profile.cpiScore > 65) priorities.push("Khẩn cấp phòng vệ rủi ro vận hạn (CPI)");
    if (cbiNorm < 40 || cbiNorm > 70) priorities.push("Cần can thiệp di cư vi khí hậu (CBI)");
    if (profile.wesScore < 50) priorities.push("Cần điều chỉnh định vị nghề nghiệp khớp Tố công (WES)");
    if (profile.fhiScore < 50) priorities.push("Cần hàn gắn mâu thuẫn cung phu thê (FHI)");

    return {
      gufiScore: finalGufi,
      rankLevel: rank,
      interventionPriority: priorities,
      recommendation: `Mệnh chủ đạt cấp bậc ${rank} (${finalGufi}/100 điểm GUFI). Cần ưu tiên: ${priorities.join("; ") || "Duy trì tu tâm tích đức."}`
    };
  }
}
```
